%                     MATH SHORTCUTS
%
% This section defines custom commands for common mathematical
% notation. Instead of repeatedly writing long LaTeX commands,

%

%   Instead of: \mathbb{R}
%   Write:      \R

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%                 NUMBER SETS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Real numbers
\newcommand{\R}{\mathbb{R}}

% Complex numbers
\newcommand{\C}{\mathbb{C}}

% Natural numbers
\newcommand{\N}{\mathbb{N}}

% Integers
\newcommand{\Z}{\mathbb{Z}}

% Rational numbers
\newcommand{\Q}{\mathbb{Q}}

% Prime numbers
\newcommand{\Pp}{\mathbb{P}}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%                 LINEAR ALGEBRA
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Transpose
% Usage: A^\top
\newcommand{\trans}{^\top}

% Rank of a matrix
\newcommand{\rank}{\operatorname{rank}}

% Kernel/null space
\newcommand{\kernel}{\operatorname{ker}}

% Image/column space
\newcommand{\image}{\operatorname{im}}

% Trace
\newcommand{\trace}{\operatorname{tr}}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%                 CALCULUS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Differential operator
% Usage: \dd x
% Produces: dx with proper spacing
\newcommand{\dd}{\,\mathrm{d}}

% Partial derivative
% Usage:
% \pdv{f}{x}
% gives ∂f/∂x
\newcommand{\pdv}[2]{\frac{\partial #1}{\partial #2}}

% Gradient
\newcommand{\grad}{\nabla}

% Laplacian
\newcommand{\laplace}{\Delta}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%                 ALGEBRA / ANALYSIS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Absolute value
% Usage: \abs{x}
\newcommand{\abs}[1]{\left|#1\right|}

% Norm
% Usage: \norm{x}
\newcommand{\norm}[1]{\left\|#1\right\|}

% Inner product
% Usage: \inner{x}{y}
\newcommand{\inner}[2]{\left\langle #1,#2 \right\rangle}

% Floor function
\newcommand{\floor}[1]{\left\lfloor #1 \right\rfloor}

% Ceiling function
\newcommand{\ceil}[1]{\left\lceil #1 \right\rceil}

% Function notation
% Usage:
% \func{f}{x}
% gives f(x)
\newcommand{\func}[2]{#1(#2)}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%                 SET THEORY
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Set builder notation
% Usage:
% \set{x \in \R}
% gives {x ∈ R}
\newcommand{\set}[1]{\left\{#1\right\}}

% Empty set
\newcommand{\emptysetc}{\varnothing}

% Cardinality
% Usage: \card(A)
\newcommand{\card}[1]{\left|#1\right|}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Common limits

\newdimen\boxfigwidth

\newenvironment{bigbox}
{
		\begingroup
		\boxfigwidth=\hsize
		\advance\boxfigwidth by -10\fboxrule
		\advance\boxfigwidth by -10\fboxsep
		\setbox4=\vbox\bgroup
		\hsize\boxfigwidth
		\hrule height0pt width\boxfigwidth
		\smallskip
		\linewidth=\boxfigwidth
}
{
		\smallskip
		\egroup
		\fbox{\box4}
		\endgroup
}
